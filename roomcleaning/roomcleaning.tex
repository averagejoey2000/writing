\item \textbf{Procedure}
  \begin{todolist}
  \item Overheads
  \begin{todolist}
\item Clean Lights
\item Dust Vent
\end{todolist}
  \item Desk
  \begin{todolist}
  \item  Remove all items from drawers.
  \item  Wipe All Surfaces with Multipurpose Cleaner Spray and Rag
  \item  Disconnect All Electronic Equipment Except Telephone
  \end{todolist}
  \item Cabinet
  \begin{todolist}
  \item  Remove all items from drawers.
  \item  Wipe All Surfaces with Multipurpose Cleaner Spray and Rag
  \end{todolist}
  \item Defrost and Clean Fridge
  \item Dress Bed with Sheets and Pillows if available
  \item Bathroom
  \begin{todolist}
  \item Scrub Toilet Bowl with bowl brush and Vacuum Toilet Bowl Cleaner.
  \item Wipe toilet surfaces with Multipurpose Spray.
  \item Clean shower floor with bleach, scrub with abrasive.
  \item Clear any hair from drains.
  \item Clean Sink with sponge and Revive
  \item Empty Medicine Cabinet.
  \item Wipe Medicine Cabinet Door Mirror with Glass Cleaner and Rag.
  \end{todolist}
  \item Bulkheads
  \begin{todolist}
  \item Wipe Mirrors and Window with Glass Cleaner and Rag.
  \item Wipe walls with Multipurpose Spray and Rag.
  \end{todolist}
  \item Floor
  \begin{todolist}
  \item Remove all garbage from room.
  \item Vacuum carpet.
  \end{todolist}
  \end{todolist}
  \end{itemize}
